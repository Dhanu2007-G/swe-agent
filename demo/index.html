# src/agent/graph.py — LangGraph state machine
from langgraph.graph import StateGraph, END
from src.agent.state import AgentState

def build_graph(checkpointer) → CompiledGraph:
  graph = StateGraph(AgentState)
  graph.add_node("read_issue", nodes.read_issue)
  graph.add_node("plan", nodes.plan)
  graph.add_node("code", nodes.code)
  graph.add_node("test", nodes.test)
  graph.add_node("correct", nodes.correct)
  graph.add_node("open_pr", nodes.open_pr)
  graph.add_node("fail", nodes.fail)

  graph.set_entry_point("read_issue")
  graph.add_conditional_edges("read_issue", route_after_read)
  graph.add_edge("plan", "code")
  graph.add_edge("code", "test")
  graph.add_conditional_edges("test", route_after_test)
  graph.add_edge("correct", "test")

  return graph.compile(checkpointer=checkpointer)

Quick Start — Deploy in 5 Minutes

# 1. Clone and configure $ git clone https://github.com/you/swe-agent $ cp .env.template .env # Edit .env: set ANTHROPIC_API_KEY, GITHUB_TOKEN, GITHUB_WEBHOOK_SECRET, DATABASE_URL
# 2. Run database migrations $ DATABASE_URL=postgresql+asyncpg://... alembic upgrade head
# 3. Start all services $ docker compose up -d ✓ api started http://localhost:8000 ✓ worker started (RQ worker process) ✓ redis started redis://localhost:6379 ✓ postgres started postgresql://localhost:5432
# 4. Verify health $ curl http://localhost:8000/health {"status":"ok","redis":"ok","db":"ok","version":"1.0.0"}
# 5. Configure GitHub webhook # → repo Settings → Webhooks → Add webhook # → URL: https://your-domain.com/webhooks/github # → Content-type: application/json # → Events: Issues only
# 6. Add 'agent-fix' label to any GitHub issue → agent runs! 🚀
